import { createClient } from '@supabase/supabase-js'

const supabaseAdmin = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_SERVICE_ROLE_KEY
)

export default async function handler(req, res) {
  if (req.method !== 'POST') return res.status(405).json({ error: 'Method not allowed' })

  // --- Verify caller is a superadmin ---
  const authHeader = req.headers.authorization || ''
  const token = authHeader.replace('Bearer ', '').trim()
  if (!token) return res.status(401).json({ error: 'Missing auth token' })

  const { data: { user }, error: userErr } = await supabaseAdmin.auth.getUser(token)
  if (userErr || !user) return res.status(401).json({ error: 'Invalid token' })

  const { data: profile } = await supabaseAdmin
    .from('profiles')
    .select('role')
    .eq('id', user.id)
    .single()
  if (profile?.role !== 'superadmin') {
    return res.status(403).json({ error: 'Only superadmins can register officers' })
  }

  // --- Proceed ---
  const { full_name, email, phone, password } = req.body
  if (!full_name || !email || !password) {
    return res.status(400).json({ error: 'full_name, email and password are required' })
  }

  const { data: authUser, error: authErr } = await supabaseAdmin.auth.admin.createUser({
    email,
    password,
    email_confirm: true,
  })
  if (authErr) return res.status(400).json({ error: authErr.message })

  const { error: offErr } = await supabaseAdmin.from('officers').insert({
    full_name,
    email: email.toLowerCase(),
    phone: phone || null,
    status: 'active',
  })
  if (offErr) return res.status(400).json({ error: offErr.message })

  const { error: profErr } = await supabaseAdmin.from('profiles').insert({
    id: authUser.user.id,
    full_name,
    email,
    role: 'officer',
  })
  if (profErr) return res.status(400).json({ error: profErr.message })

  res.status(200).json({ ok: true })
}

export const features = {
  camera: typeof navigator !== 'undefined' && !!(navigator.mediaDevices && navigator.mediaDevices.getUserMedia),
  webauthn: typeof window !== 'undefined' && !!(window.PublicKeyCredential),
  haptics: typeof navigator !== 'undefined' && typeof navigator.vibrate === 'function',
  secureContext: typeof window !== 'undefined' && (window.isSecureContext === true || window.location.hostname === 'localhost'),
}

export async function webauthnAvailable() {
  if (!features.webauthn) return false
  try {
    return await PublicKeyCredential.isUserVerifyingPlatformAuthenticatorAvailable()
  } catch {
    return false
  }
}

export function isIOS() {
  if (typeof navigator === 'undefined') return false
  return /iPad|iPhone|iPod/.test(navigator.userAgent) ||
    (navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1)
}

export function isMobile() {
  if (typeof navigator === 'undefined') return false
  return /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent)
}

import { useEffect, useMemo, useState } from 'react'
import { useNavigate } from 'react-router-dom'
import {
  LineChart, Line, BarChart, Bar, PieChart, Pie, Cell,
  XAxis, YAxis, Tooltip, ResponsiveContainer, CartesianGrid,
} from 'recharts'
import { supabase } from '../lib/supabase'
import Logo from '../components/Logo'

const COLORS = ['#006600', '#BB0000', '#8A8A80']
const CHART_H = 200

export default function AdminDashboard() {
  const navigate = useNavigate()
  const [email, setEmail] = useState('')
  const [officers, setOfficers] = useState([])
  const [voters, setVoters] = useState([])
  const [loading, setLoading] = useState(true)
  const [search, setSearch] = useState('')
  const [sort, setSort] = useState({ key: 'created_at', dir: 'desc' })

  useEffect(() => {
    (async () => {
      const { data: { user } } = await supabase.auth.getUser()
      setEmail(user?.email ?? '')

      const [{ data: off }, { data: vot }] = await Promise.all([
        supabase.from('officers').select('*').order('created_at', { ascending: false }),
        supabase.from('voters').select('id, county, gender, created_at, verified_at'),
      ])

      console.log('[dashboard] officers:', off?.length, 'voters:', vot?.length)
      setOfficers(off ?? [])
      setVoters(vot ?? [])
      setLoading(false)
    })()
  }, [])

  async function handleLogout() {
    await supabase.auth.signOut()
    navigate('/login', { replace: true })
  }

  const filteredOfficers = useMemo(() => {
    let list = [...officers]
    if (search.trim()) {
      const q = search.toLowerCase()
      list = list.filter(o =>
        o.full_name?.toLowerCase().includes(q) ||
        o.email?.toLowerCase().includes(q) ||
        o.phone?.includes(q)
      )
    }
    list.sort((a, b) => {
      const av = a[sort.key] ?? ''
      const bv = b[sort.key] ?? ''
      const cmp = av < bv ? -1 : av > bv ? 1 : 0
      return sort.dir === 'asc' ? cmp : -cmp
    })
    return list
  }, [officers, search, sort])

  const toggleSort = (key) => setSort(s =>
    s.key === key ? { key, dir: s.dir === 'asc' ? 'desc' : 'asc' } : { key, dir: 'asc' }
  )

  const todayStr = new Date().toDateString()
  const todayCount = voters.filter(v => new Date(v.created_at).toDateString() === todayStr).length
  const verifiedToday = voters.filter(v => v.verified_at && new Date(v.verified_at).toDateString() === todayStr).length

  const genderData = useMemo(() => {
    const map = { male: 0, female: 0, other: 0 }
    voters.forEach(v => { if (map[v.gender] !== undefined) map[v.gender]++ })
    return [
      { name: 'Male', value: map.male },
      { name: 'Female', value: map.female },
      { name: 'Other', value: map.other },
    ].filter(d => d.value > 0)
  }, [voters])

  const countyData = useMemo(() => {
    const map = {}
    voters.forEach(v => { map[v.county] = (map[v.county] || 0) + 1 })
    return Object.entries(map)
      .map(([county, count]) => ({ county, count }))
      .sort((a, b) => b.count - a.count)
      .slice(0, 8)
  }, [voters])

  const dailyData = useMemo(() => {
    const days = []
    for (let i = 13; i >= 0; i--) {
      const d = new Date()
      d.setDate(d.getDate() - i)
      const key = d.toDateString()
      days.push({
        day: d.toLocaleDateString('en-KE', { month: 'short', day: 'numeric' }),
        count: voters.filter(v => new Date(v.created_at).toDateString() === key).length,
      })
    }
    return days
  }, [voters])

  const hasVoters = voters.length > 0

  return (
    <div className="admin">
      <header className="admin-header">
        <Logo size={36} showText />
        <div className="admin-header-right">
          <span className="admin-user">{email}</span>
          <button className="btn-ghost" onClick={handleLogout}>Sign out</button>
        </div>
      </header>

      <main className="admin-main">
        <div className="admin-title-row">
          <div>
            <h1>Superadmin Dashboard</h1>
            <p className="muted">Live overview of registration &amp; verification</p>
          </div>
          <div style={{ display: 'flex', gap: 8, flexWrap: 'wrap' }}>
            <button className="btn-ghost" onClick={() => navigate('/admin/verify')}>Verify Voter</button>
            <button className="btn-ghost" onClick={() => navigate('/admin/voter-registration')}>Register Voter</button>
            <button className="btn-primary" onClick={() => navigate('/admin/register-officer')}>+ Register Officer</button>
          </div>
        </div>

        <section className="kpi-grid kpi-grid-6">
          <div className="kpi-card"><span className="kpi-label">Total Voters</span><span className="kpi-value">{voters.length}</span></div>
          <div className="kpi-card"><span className="kpi-label">Registered Today</span><span className="kpi-value">{todayCount}</span></div>
          <div className="kpi-card"><span className="kpi-label">Verified Today</span><span className="kpi-value">{verifiedToday}</span></div>
          <div className="kpi-card"><span className="kpi-label">Total Officers</span><span className="kpi-value">{officers.length}</span></div>
          <div className="kpi-card"><span className="kpi-label">Active Officers</span><span className="kpi-value">{officers.filter(o => o.status === 'active').length}</span></div>
          <div className="kpi-card"><span className="kpi-label">System Status</span><span className="kpi-value kpi-value-status">Online</span></div>
        </section>

        {!hasVoters ? (
          <div className="chart-empty-card">
            <h3>No data to visualise yet</h3>
            <p className="muted">Charts will appear once voters are registered.</p>
            <button className="btn-primary" onClick={() => navigate('/admin/voter-registration')}>
              Register your first voter 
            </button>
          </div>
        ) : (
          <>
            <section className="charts-grid">
              <div className="chart-card chart-wide">
                <h3>Registrations  Last 14 Days</h3>
                <div style={{ width: '100%', height: CHART_H }}>
                  <ResponsiveContainer width="100%" height="100%">
                    <LineChart data={dailyData}>
                      <CartesianGrid strokeDasharray="3 3" stroke="#EBEBE6" />
                      <XAxis dataKey="day" tick={{ fontSize: 11 }} />
                      <YAxis allowDecimals={false} tick={{ fontSize: 11 }} />
                      <Tooltip />
                      <Line type="monotone" dataKey="count" stroke="#006600" strokeWidth={2.5} dot={{ r: 3 }} />
                    </LineChart>
                  </ResponsiveContainer>
                </div>
              </div>

              <div className="chart-card">
                <h3>Gender Split</h3>
                <div style={{ width: '100%', height: CHART_H }}>
                  <ResponsiveContainer width="100%" height="100%">
                    <PieChart>
                      <Pie data={genderData} dataKey="value" nameKey="name" cx="50%" cy="50%" outerRadius={70} label>
                        {genderData.map((_, i) => <Cell key={i} fill={COLORS[i % COLORS.length]} />)}
                      </Pie>
                      <Tooltip />
                    </PieChart>
                  </ResponsiveContainer>
                </div>
              </div>
            </section>

            <section className="charts-grid">
              <div className="chart-card chart-wide">
                <h3>Voters by County (Top 8)</h3>
                <div style={{ width: '100%', height: CHART_H }}>
                  <ResponsiveContainer width="100%" height="100%">
                    <BarChart data={countyData} layout="vertical" margin={{ left: 20 }}>
                      <CartesianGrid strokeDasharray="3 3" stroke="#EBEBE6" />
                      <XAxis type="number" allowDecimals={false} tick={{ fontSize: 11 }} />
                      <YAxis dataKey="county" type="category" width={90} tick={{ fontSize: 11 }} />
                      <Tooltip />
                      <Bar dataKey="count" fill="#006600" radius={[0, 4, 4, 0]} />
                    </BarChart>
                  </ResponsiveContainer>
                </div>
              </div>
            </section>
          </>
        )}

        <section className="officers-section">
          <div className="officers-toolbar">
            <h2>Registration Officers</h2>
            <input type="search" className="search-input"
              placeholder="Search by name, email, or phone"
              value={search} onChange={(e) => setSearch(e.target.value)} />
          </div>

          {loading ? (
            <div className="skeleton-list">
              {[...Array(3)].map((_, i) => <div key={i} className="skeleton-row" />)}
            </div>
          ) : filteredOfficers.length === 0 ? (
            <div className="empty-state">
              <p>No officers {search ? 'matching your search' : 'yet'}.</p>
              <button className="btn-primary" onClick={() => navigate('/admin/register-officer')}>
                Register {search ? 'another' : 'the first'} officer
              </button>
            </div>
          ) : (
            <div className="table-wrap">
              <table className="officers-table">
                <thead>
                  <tr>
                    <th onClick={() => toggleSort('full_name')} className={sort.key === 'full_name' ? 'sorted' : ''}>
                      Full Name {sort.key === 'full_name' && (sort.dir === 'asc' ? '' : '')}
                    </th>
                    <th onClick={() => toggleSort('email')} className={sort.key === 'email' ? 'sorted' : ''}>
                      Email {sort.key === 'email' && (sort.dir === 'asc' ? '' : '')}
                    </th>
                    <th>Phone</th>
                    <th>Status</th>
                    <th onClick={() => toggleSort('created_at')} className={sort.key === 'created_at' ? 'sorted' : ''}>
                      Registered {sort.key === 'created_at' && (sort.dir === 'asc' ? '' : '')}
                    </th>
                  </tr>
                </thead>
                <tbody>
                  {filteredOfficers.map(o => (
                    <tr key={o.id}>
                      <td data-label="Name">{o.full_name}</td>

  )
}

import { useState } from 'react'
import { useNavigate } from 'react-router-dom'
import { supabase } from '../lib/supabase'
import Logo from '../components/Logo'

export default function Login() {
  const navigate = useNavigate()
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  async function handleSubmit(e) {
    e.preventDefault()
    setError('')
    setLoading(true)

    console.log('[login] attempting:', email)

    const { data: auth, error: signInErr } = await supabase.auth.signInWithPassword({ email, password })

    if (signInErr) {
      console.error('[login] sign-in error:', signInErr.message)
      setLoading(false)
      return setError(signInErr.message)
    }

    console.log('[login] sign-in OK, user.id=', auth.user.id)

    const { data: profile, error: pErr } = await supabase
      .from('profiles')
      .select('role')
      .eq('id', auth.user.id)
      .maybeSingle()

    console.log('[login] profile fetch  error:', pErr?.message ?? 'none', ' data:', profile)

    setLoading(false)

    if (pErr) {
      await supabase.auth.signOut()
      return setError('Profile lookup failed: ' + pErr.message)
    }
    if (!profile) {
      await supabase.auth.signOut()
      return setError('No profile row for this account. Contact the administrator.')
    }

    const dest = profile.role === 'superadmin' ? '/admin' : '/officer'
    console.log('[login] DECISION  role:', profile.role, ' dest:', dest)
    navigate(dest, { replace: true })
  }

  return (
    <div className="login-page">
      <div className="login-brand">
        <div className="login-brand-inner">
          <Logo size={64} />
          <h1>Independent Electoral<br/>and Boundaries Commission</h1>
          <p>Voter Registration &amp; Verification System</p>
          <div className="brand-stripes"><span/><span/><span/></div>
        </div>
      </div>

      <div className="login-form-wrap">
        <form className="login-form" onSubmit={handleSubmit}>
          <h2>Sign In</h2>
          <p className="muted">IEBC staff  admins &amp; registration officers</p>

          <label>
            Email
            <input type="email" required autoComplete="email"
              value={email} onChange={(e) => setEmail(e.target.value)}
              placeholder="you@iebc.local" />
          </label>

          <label>
            Password
            <input type="password" required autoComplete="current-password"
              value={password} onChange={(e) => setPassword(e.target.value)}
              placeholder="" />
          </label>

          {error && <div className="form-error" role="alert">{error}</div>}

          <button type="submit" className="btn-primary" disabled={loading}>
            {loading ? 'Signing in' : 'Sign In'}
          </button>

          <p className="login-footer">Protected by IEBC Security Policy</p>
        </form>
      </div>
    </div>
  )
}

import { useState } from 'react'
import { useNavigate } from 'react-router-dom'
import { supabase } from '../lib/supabase'
import Logo from '../components/Logo'

export default function RegisterOfficer() {
  const navigate = useNavigate()
  const [form, setForm] = useState({ full_name: '', email: '', phone: '' })
  const [saving, setSaving] = useState(false)
  const [error, setError] = useState('')
  const [success, setSuccess] = useState(false)

  const update = (k, v) => setForm(f => ({ ...f, [k]: v }))

  async function handleSubmit(e) {
    e.preventDefault()
    setError('')
    setSaving(true)
    const { data: { session } } = await supabase.auth.getSession()
    const res = await fetch('/api/create-officer', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${session?.access_token ?? ''}`,
      },
      body: JSON.stringify({
        full_name: form.full_name.trim(),
        email: form.email.trim().toLowerCase(),
        phone: form.phone.trim() || null,
        password: 'IEBC@2026',
      }),
    })
    const data = await res.json()
    setSaving(false)
    if (!res.ok) return setError(data.error || 'Failed to create officer')
    setSuccess(true)
    setTimeout(() => navigate('/admin'), 1400)
  }

  if (success) {
    return (
      <div className="success-screen">
        <svg viewBox="0 0 100 100" width="120" height="120" className="success-tick">
          <circle cx="50" cy="50" r="42" fill="none" stroke="#006600" strokeWidth="3" className="tick-circle" />
          <path d="M30 52 L45 67 L72 38" fill="none" stroke="#006600" strokeWidth="6"
            strokeLinecap="round" strokeLinejoin="round" className="tick-check" />
        </svg>
        <h2>Officer Registered</h2>
        <p className="muted">Redirecting to dashboard</p>
      </div>
    )
  }

  return (
    <div className="register-page">
      <header className="admin-header">
        <Logo size={32} showText />
        <button className="btn-ghost" onClick={() => navigate('/admin')}> Back</button>
      </header>

      <main className="register-main">
        <div className="form-card">
          <h1>Register Officer</h1>
          <p className="muted">Add a new voter registration officer to the system</p>

          <form onSubmit={handleSubmit} className="register-form">
            <label>
              Full Name *
              <input required value={form.full_name}
                onChange={(e) => update('full_name', e.target.value)}
                placeholder="Jane Wanjiku Kamau" />
            </label>
            <label>
              Email *
              <input required type="email" value={form.email}
                onChange={(e) => update('email', e.target.value)}
                placeholder="jane.kamau@iebc.go.ke" />
            </label>
            <label>
              Phone
              <input type="tel" value={form.phone}
                onChange={(e) => update('phone', e.target.value)}
                placeholder="+254 700 000 000" />
            </label>

            {error && <div className="form-error" role="alert">{error}</div>}

            <div className="form-actions">
              <button type="button" className="btn-ghost" onClick={() => navigate('/admin')}>Cancel</button>
              <button type="submit" className="btn-primary" disabled={saving}>
                {saving ? 'Registering' : 'Register Officer'}
              </button>
            </div>
          </form>
        </div>
      </main>
    </div>
  )
}

import { useState, useRef, useEffect } from 'react'
import { useNavigate } from 'react-router-dom'
import { supabase } from '../lib/supabase'
import { LOCATIONS } from '../lib/locations'
import { features } from '../lib/features'
import Logo from '../components/Logo'

const STEPS = ['Personal', 'Location', 'Biometric']

export default function VoterRegistration() {
  const navigate = useNavigate()
  const [step, setStep] = useState(0)
  const [form, setForm] = useState({
    first_name: '', middle_name: '', last_name: '',
    gender: '', id_type: 'national_id', id_number: '', dob: '',
    county: '', constituency: '', ward: '', polling_station: '',
    face_photo_data: '',
  })
  const [saving, setSaving] = useState(false)
  const [error, setError] = useState('')
  const [success, setSuccess] = useState(false)
  const [cameraOn, setCameraOn] = useState(false)

  const videoRef = useRef(null)
  const canvasRef = useRef(null)
  const streamRef = useRef(null)

  const update = (k, v) => setForm(f => ({ ...f, [k]: v }))
  const setCounty = (v) => setForm(f => ({ ...f, county: v, constituency: '', ward: '', polling_station: '' }))
  const setConstituency = (v) => setForm(f => ({ ...f, constituency: v, ward: '', polling_station: '' }))
  const setWard = (v) => setForm(f => ({ ...f, ward: v, polling_station: '' }))

  const counties = Object.keys(LOCATIONS)
  const constituencies = form.county ? Object.keys(LOCATIONS[form.county] || {}) : []
  const wards = form.county && form.constituency ? Object.keys(LOCATIONS[form.county]?.[form.constituency] || {}) : []
  const stations = form.ward ? (LOCATIONS[form.county]?.[form.constituency]?.[form.ward] || []) : []

  const maxDob = new Date()
  maxDob.setFullYear(maxDob.getFullYear() - 18)
  const maxDobStr = maxDob.toISOString().split('T')[0]

  async function startCamera() {
    setError('')
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: 'user', width: { ideal: 640 }, height: { ideal: 480 } }
      })
      streamRef.current = stream
      if (videoRef.current) {
        videoRef.current.srcObject = stream
        await videoRef.current.play()
      }
      setCameraOn(true)
    } catch (e) {
      const msg = e?.name === 'NotAllowedError'
        ? 'Camera permission denied. Allow camera access in your browser settings, then try again.'
        : e?.name === 'NotFoundError'
        ? 'No camera found on this device. Contact the ICT officer.'
        : e?.name === 'NotReadableError'
        ? 'Camera is in use by another app. Close it and try again.'
        : 'Camera error: ' + (e?.message || 'unknown')
      setError(msg)
    }
  }

  function stopCamera() {
    if (streamRef.current) {
      streamRef.current.getTracks().forEach(t => t.stop())
      streamRef.current = null
    }
    setCameraOn(false)
  }

  function captureFace() {
    if (!videoRef.current || !canvasRef.current) return
    const v = videoRef.current, c = canvasRef.current
    c.width = v.videoWidth
    c.height = v.videoHeight
    c.getContext('2d').drawImage(v, 0, 0)
    update('face_photo_data', c.toDataURL('image/jpeg', 0.85))
    stopCamera()
  }

  useEffect(() => () => stopCamera(), [])

  const canNext1 = form.first_name.trim() && form.last_name.trim() && form.gender
    && form.id_type && form.id_number.trim() && form.dob
  const canNext2 = form.county && form.constituency && form.ward && form.polling_station
  const canSubmit = canNext1 && canNext2 && form.face_photo_data

  async function handleSubmit() {
    if (!canSubmit) return
    setSaving(true); setError('')
    const { data: { user } } = await supabase.auth.getUser()
    const { error } = await supabase.from('voters').insert({
      first_name: form.first_name.trim(),
      middle_name: form.middle_name.trim() || null,
      last_name: form.last_name.trim(),
      gender: form.gender,
      id_type: form.id_type,
      id_number: form.id_number.trim(),
      dob: form.dob,
      county: form.county,
      constituency: form.constituency,
      ward: form.ward,
      polling_station: form.polling_station,
      face_photo_data: form.face_photo_data,
      registered_by: user?.id ?? null,
    })
    setSaving(false)
    if (error) return setError(error.message)
    setSuccess(true)
    setTimeout(() => navigate('/admin'), 1800)
  }

  if (success) {
    return (
      <div className="success-screen">
        <svg viewBox="0 0 100 100" width="120" height="120" className="success-tick">
          <circle cx="50" cy="50" r="42" fill="none" stroke="#006600" strokeWidth="3" className="tick-circle" />
          <path d="M30 52 L45 67 L72 38" fill="none" stroke="#006600" strokeWidth="6"
            strokeLinecap="round" strokeLinejoin="round" className="tick-check" />
        </svg>
        <h2>Voter Registered</h2>
        <p className="muted">Redirecting to dashboard</p>
      </div>
    )
  }

  return (
    <div className="register-page">
      <header className="admin-header">
        <Logo size={32} showText />
        <button className="btn-ghost" onClick={() => navigate('/admin')}> Back</button>
      </header>

      <main className="register-main">
        <div className="stepper">
          {STEPS.map((s, i) => (
            <div key={s} className={`step ${i === step ? 'active' : ''} ${i < step ? 'done' : ''}`}>
              <span className="step-num">{i < step ? '' : i + 1}</span>
              <span className="step-label">{s}</span>
            </div>
          ))}
        </div>

        <div className="form-card">
          {step === 0 && (
            <>
              <h1>Personal Details</h1>
              <p className="muted">Step 1 of 3</p>
              <div className="register-form">
                <label>First Name *<input value={form.first_name} onChange={e => update('first_name', e.target.value)} placeholder="Jane" /></label>
                <label>Middle Name<input value={form.middle_name} onChange={e => update('middle_name', e.target.value)} placeholder="Wanjiku" /></label>
                <label>Last Name *<input value={form.last_name} onChange={e => update('last_name', e.target.value)} placeholder="Kamau" /></label>

                <label>Gender *
                  <div className="radio-group">
                    {['male','female','other'].map(g => (
                      <label key={g} className="radio-pill">
                        <input type="radio" name="gender" value={g} checked={form.gender === g} onChange={e => update('gender', e.target.value)} />
                        <span>{g.charAt(0).toUpperCase() + g.slice(1)}</span>
                      </label>
                    ))}
                  </div>
                </label>

                <label>ID Type *
                  <div className="radio-group">
                    <label className="radio-pill">
                      <input type="radio" name="id_type" value="national_id" checked={form.id_type === 'national_id'} onChange={e => update('id_type', e.target.value)} />
                      <span>National ID</span>
                    </label>
                    <label className="radio-pill">
                      <input type="radio" name="id_type" value="passport" checked={form.id_type === 'passport'} onChange={e => update('id_type', e.target.value)} />
                      <span>Passport</span>
                    </label>
                  </div>
                </label>

                <label>{form.id_type === 'passport' ? 'Passport Number' : 'National ID Number'} *
                  <input value={form.id_number} onChange={e => update('id_number', e.target.value)} placeholder="12345678" />
                </label>

                <label>Date of Birth *
                  <input type="date" max={maxDobStr} value={form.dob} onChange={e => update('dob', e.target.value)} />
                </label>
              </div>
            </>
          )}

          {step === 1 && (
            <>
              <h1>Location & Polling Station</h1>
              <p className="muted">Step 2 of 3</p>
              <div className="register-form">
                <label>County *
                  <select value={form.county} onChange={e => setCounty(e.target.value)}>
                    <option value="">Select county</option>
                    {counties.map(c => <option key={c} value={c}>{c}</option>)}
                  </select>
                </label>

                <label>Constituency *
                  <select value={form.constituency} onChange={e => setConstituency(e.target.value)} disabled={!form.county}>
                    <option value="">{form.county ? 'Select constituency' : 'Select a county first'}</option>
                    {constituencies.map(c => <option key={c} value={c}>{c}</option>)}
                  </select>
                </label>

                <label>Ward (CAW) *
                  <select value={form.ward} onChange={e => setWard(e.target.value)} disabled={!form.constituency}>
                    <option value="">{form.constituency ? 'Select ward' : 'Select a constituency first'}</option>
                    {wards.map(w => <option key={w} value={w}>{w}</option>)}
                  </select>
                </label>

                <label>Polling Station *
                  <select value={form.polling_station} onChange={e => update('polling_station', e.target.value)} disabled={!form.ward}>
                    <option value="">{form.ward ? 'Select polling station' : 'Select a ward first'}</option>
                    {stations.map(p => <option key={p} value={p}>{p}</option>)}
                  </select>
                </label>
              </div>
            </>
          )}

          {step === 2 && (
            <>
              <h1>Biometric Capture</h1>
              <p className="muted">Step 3 of 3</p>
              <div className="biometric-section">
                <h3>Face Capture</h3>

                {!features.camera && (
                  <div className="form-error">
                    This browser doesn't support camera access. Use Chrome, Safari 15+, Edge, or Firefox on a modern phone or laptop.
                  </div>
                )}

                {features.camera && form.face_photo_data && (
                  <div className="face-preview">
                    <img src={form.face_photo_data} alt="Captured face" />
                    <button className="btn-ghost" onClick={() => { update('face_photo_data', ''); startCamera() }}>Retake</button>
                  </div>
                )}

                {features.camera && !form.face_photo_data && cameraOn && (
                  <div className="camera-view">
                    <video ref={videoRef} autoPlay playsInline muted className="camera-video" />
                    <canvas ref={canvasRef} style={{ display: 'none' }} />
                    <div className="camera-actions">
                      <button className="btn-primary" onClick={captureFace}>Capture</button>
                      <button className="btn-ghost" onClick={stopCamera}>Cancel</button>
                    </div>
                  </div>
                )}

                {features.camera && !form.face_photo_data && !cameraOn && (
                  <button className="btn-primary" onClick={startCamera}>Start Camera</button>
                )}

                <h3 style={{ marginTop: 24 }}>Fingerprint Enroll</h3>
                <p className="muted">WebAuthn enrollment lands in the next step.</p>
              </div>
            </>
          )}

          {error && <div className="form-error" style={{ marginTop: 16 }}>{error}</div>}

          <div className="form-actions" style={{ marginTop: 24 }}>
            {step > 0 && <button className="btn-ghost" onClick={() => setStep(s => s - 1)}> Back</button>}
            {step < 2 && (
              <button className="btn-primary"
                disabled={step === 0 ? !canNext1 : !canNext2}
                onClick={() => setStep(s => s + 1)}>Continue </button>
            )}
            {step === 2 && (
              <button className="btn-primary" disabled={!canSubmit || saving} onClick={handleSubmit}>
                {saving ? 'Registering' : 'Complete Registration'}
              </button>
            )}
          </div>
        </div>
      </main>
    </div>
  )
}

import { useEffect, useState } from 'react'
import { BrowserRouter, Routes, Route, Navigate } from 'react-router-dom'
import { supabase } from './lib/supabase'
import Login from './pages/Login'
import AdminDashboard from './pages/AdminDashboard'
import RegisterOfficer from './pages/RegisterOfficer'
import VoterRegistration from './pages/VoterRegistration'
import OfficerDashboard from './pages/OfficerDashboard'

function useSession() {
  const [session, setSession] = useState(undefined)
  useEffect(() => {
    supabase.auth.getSession().then(({ data }) => setSession(data.session))
    const { data: sub } = supabase.auth.onAuthStateChange((_e, s) => setSession(s))
    return () => sub.subscription.unsubscribe()
  }, [])
  return session
}

function useRole(session) {
  const [role, setRole] = useState(undefined)
  useEffect(() => {
    if (session === undefined) return
    if (!session) { setRole(null); return }
    console.log('[guard] fetching role for', session.user.email)
    supabase.from('profiles').select('role').eq('id', session.user.id).maybeSingle()
      .then(({ data, error }) => {
        console.log('[guard] role result  error:', error?.message ?? 'none', ' data:', data)
        if (error || !data) setRole('__invalid__')
        else setRole(data.role)
      })
  }, [session])
  return role
}

function Protected({ children, adminOnly = false }) {
  const session = useSession()
  const role = useRole(session)

  if (session === undefined) return <div className="boot-loader">Loading</div>
  if (!session) return <Navigate to="/login" replace />
  if (role === undefined) return <div className="boot-loader">Loading</div>

  if (adminOnly && role !== 'superadmin') {
    console.log('[guard] adminOnly blocked  role:', role, ' /officer')
    return <Navigate to="/officer" replace />
  }
  console.log('[guard] allowing  role:', role)
  return children
}

export default function App() {
  return (
    <BrowserRouter>
      <Routes>
        <Route path="/login" element={<Login />} />
        <Route path="/admin" element={<Protected adminOnly><AdminDashboard /></Protected>} />
        <Route path="/admin/register-officer" element={<Protected adminOnly><RegisterOfficer /></Protected>} />
        <Route path="/admin/voter-registration" element={<Protected><VoterRegistration /></Protected>} />
        <Route path="/officer" element={<Protected><OfficerDashboard /></Protected>} />
        <Route path="*" element={<Navigate to="/login" replace />} />
      </Routes>
    </BrowserRouter>
  )
}

========== .\README.md ==========
# React + Vite

This template provides a minimal setup to get React working in Vite with HMR and some Oxlint rules.

Currently, two official plugins are available:

- [@vitejs/plugin-react](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react) uses [Oxc](https://oxc.rs)
- [@vitejs/plugin-react-swc](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react-swc) uses [SWC](https://swc.rs/)

## React Compiler

The React Compiler is not enabled on this template because of its impact on dev & build performances. To add it, see [this documentation](https://react.dev/learn/react-compiler/installation).

## Expanding the Oxlint configuration

If you are developing a production application, we recommend using TypeScript with type-aware lint rules enabled. Check out the [TS template](https://github.com/vitejs/vite/tree/main/packages/create-vite/template-react-ts) for information on how to integrate TypeScript and Oxlint's TypeScript related rules in your project.
